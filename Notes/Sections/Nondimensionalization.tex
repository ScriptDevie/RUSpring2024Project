\hline
\textbf{Quantity} & \textbf{Scale} & \textbf{New Variable}\\
\hline
\hline
$\dim{z}$                            & $l_{z}$    & $\dim{z} = l_{z} z$\\
\hline
$\dim{x}$                            & $l_{x,y}$  & $\dim{x} = l_{x,y} x$\\
\hline
$\dim{y}$                            & $l_{x,y}$  & $\dim{y} = l_{x,y} y$\\
\hline
$n(\dim{x}, \dim{y}, \dim{z})$       & $\sigma_n$ & $n(\dim{x}, \dim{y}, \dim{z}) = \sigma_n n(x,y,z)$\\
\hline
$\dim{A}(\dim{x}, \dim{y}, \dim{z})$ & $A_0$      & $\dim{A}(l_{x,y} \dim{x}, l_{x,y} \dim{y}, l_{z} \dim{z}) = A_0 A(x,y,z)$  \\
\hline
\end{tabular}
\end{center}

 \begin{equation}\label{EQ:helmholtz_equation_2}
 \begin{split}
   & \frac{1}{l_{z}^{2}} \ppder{A}{z} + \frac{1}{l_{x,y}^{2}} \nabla_{\perp}^2 A + k^2 n_0^2 A +  2 k^2 n_0 \sigma_n n(x,y,z) A + k^2 \sigma_n^2 n^2(x,y,z) A = 0 \\
   &\quad  \mbox{for}
   \quad -\infty < x,y < \infty, \quad 0 < z < \frac{L}{L_0} \quad  \mbox{where} \quad A(x,y,0) = A_{i}(x,y)\exp{i \Phi(x,y)}
 \end{split}
 \end{equation}
 and $A_{i}(x,y)$ is of finite support, \ie $A_{i}(x,y) = 0$ when $\sqrt{x^2 + y^2} > \frac{D_{ap}}{l_{x,y}}$.



\begin{equation}\label{EQ:helmholtz_equation_3}
\begin{split}
  & \frac{l_{x,y}^{2}}{l_{z}^{2}} \ppder{A}{z} + \nabla_{\perp}^2 A + l_{x,y}^{2} k^2 n_0^2 A 
  +  2 l_{x,y}^{2} k^2 n_0 \sigma_n n(x,y,z) A + l_{x,y}^{2} k^2 \sigma_n^2 n^2(x,y,z) A = 0 \\
  &\quad  \mbox{for}   
  \quad -\infty < x,y < \infty, \quad 0 \leq z \leq \frac{L}{L_0} \quad  
  \mbox{where} \quad A(x,y,0) = A_{i}(x,y)\exp{i \Phi(x,y)}
\end{split}
\end{equation}
and $A_{i}(x,y)$ is of finite support, \ie $A_{i}(x,y) = 0$ when $\sqrt{x^2 + y^2} > \frac{D_{ap}}{l_{x,y}}$.
