
\section{Paraxial Approximation}
Letting $A = \exp{i \frac{\beta}{\epsilon} n_0 z} U(x,y,z)$ gives
\begin{equation}\label{EQ:helmholtz_equation_5}
\begin{split}
  & \epsilon^2 \left( -\frac{\beta^2}{\epsilon^2} n_0^2 U + 2 i \frac{\beta}{\epsilon} n_0 \pder{U}{z} + \ppder{U}{z} \right) 
  + \nabla_{\perp}^2 U + \beta^2 n_0^2 U +  2 \gamma^2 n_0 n(x,y,z) U + \nu^2 n^2(x,y,z) U = 0 \\
  & 2 i \beta \epsilon n_0 \pder{U}{z} + \epsilon^2\ppder{U}{z} + \nabla_{\perp}^2 U +  2 \gamma^2 n_0 n(x,y,z) U + \nu^2 n^2(x,y,z) U = 0 \\  
  & 2 i \alpha n_0 \pder{U}{z} + \epsilon^2\ppder{U}{z} + \nabla_{\perp}^2 U +  2 \gamma^2 n_0 n(x,y,z) U + \nu^2 n^2(x,y,z) U = 0 \\
  &\quad  \mbox{for} \quad -\infty < x,y < \infty, \quad 0 \leq z \leq l_{pr}  \quad \mbox{where} \quad U(x,y,0) = U_{i}(x,y) \exp{i\Phi(x,y)}
\end{split}
\end{equation}
and $U_{i}(x,y)$ is of finite support, \ie $U_{i}(x,y) = 0$ when $\sqrt{x^2 + y^2} > d_{ap}$.

Taking the dominate balance to be $\{\nu^2 \sim \epsilon^2, \alpha \sim \gamma^2 \sim 1 \}$

\subsection{Nondimensionalized Correlation}
\begin{equation}\label{EQ:correlation_2}
  C_{cor}(\|\vect{\dim{r}}_1-\vect{\dim{r}}_{2}\|) = 1 - \frac{C_n^2}{2\sigma_n^2}
    \begin{cases}
        l_{0}^{-4/3} \|\vect{\dim{r}}_1-\vect{\dim{r}}_{2}\|^{2} & 0 < \|\vect{\dim{r}}_1-\vect{\dim{r}}_{2}\| \leq l_{0} \\
        \|\vect{\dim{r}}_1-\vect{\dim{r}}_{2}\|^{2/3} &  l_{0}  \leq \|\vect{\dim{r}}_1-\vect{\dim{r}}_{2}\| \leq  L_{0}  \\
    \end{cases}
\end{equation}


\begin{equation}\label{EQ:correlation_2}
  C_{cor}(\|\vect{r}_1-\vect{r}_{2}\|) = 1 - \frac{C_n^2}{2\sigma_n^2}
    \begin{cases}
        \frac{L_0^2}{l_{0}^{4/3}} \left( (z_{1} - z_{2})^{2} + \epsilon^2 \left|\vect{r}_{\perp,1}-\vect{r}_{\perp,2}\right|^{2}\right) & 0 \leq \sqrt{(z_{1} - z_{2})^{2} + \epsilon^2 \left|\vect{r}_{\perp,1}-\vect{r}_{\perp,2}\right|^{2}}  \leq  \epsilon \\
        L_0^{2/3} \left( (z_{1} - z_{2})^{2} + \epsilon^2 \left|\vect{r}_{\perp,1}-\vect{r}_{\perp,2}\right|^{2}\right)^{1/3} &  \epsilon \leq \sqrt{(z_{1} - z_{2})^{2} + \epsilon^2 \left|\vect{r}_{\perp,1}-\vect{r}_{\perp,2}\right|^{2}}  \leq  1   \\
   \end{cases}
\end{equation}


\begin{equation}
  C_{cor}(\|\vect{r}_1-\vect{r}_{2}\|) = 1 - \frac{1}{C_0^{2/3}}
    \begin{cases}
        \frac{L_0^{4/3}}{l_{0}^{4/3}}\left( (z_{1} - z_{2})^{2} + \epsilon^2 \left|\vect{r}_{\perp,1}-\vect{r}_{\perp,2}\right|^{2}\right) & 0 \leq \sqrt{(z_{1} - z_{2})^{2} + \epsilon^2 \left|\vect{r}_{\perp,1}-\vect{r}_{\perp,2}\right|^{2}}  \leq  \epsilon  \\
        \left( (z_{1} - z_{2})^{2} + \epsilon^2 \left|\vect{r}_{\perp,1}-\vect{r}_{\perp,2}\right|^{2}\right)^{1/3} &  \epsilon \leq \sqrt{(z_{1} - z_{2})^{2} + \epsilon^2 \left|\vect{r}_{\perp,1}-\vect{r}_{\perp,2}\right|^{2}}  \leq  1 \\
   \end{cases}
\end{equation}
