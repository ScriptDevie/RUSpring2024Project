\section{Averaging NLSE+DM}In this section we go through the averaging for NSLE+DM.\begin{equation}\begin{split}i \pder{q}{z} &+ i\frac{1}{x_p} \pder{q}{x} + \left(\avg{d} + \frac{\Del{d}(x)}{x_p}\right) \pder{q}{tt}q + s |q|^2 q  = 0, \end{split}\end{equation}We next introduce an expansion for $q(z,x,t)$ in powers of $x_p\ll1$,\begin{equation}\label{equ:CGLE+DM_MS}\begin{split}q(z,x,t) = q^{(0)}(z,x,t) + x_p q^{(1)}(z,x,t) + \dots,\end{split}\end{equation}to get the following.\paragraph{At $\O{1}$:}\begin{equation}\label{equ:CGLE+DM_O1}\begin{split}i \pder{q^{(0)}}{x} + \Del{d_r}(x)\pder{q^{(0)}}{tt}= 0 .\\\end{split}\end{equation} To solve this equation, introduce the following notation for the Fourier Transform:\begin{equation}\label{equ:Fourier_Transform}\begin{split}\F{f(t)} = \hat{f}(\omega) = \int f(t) \exp{i\omega t}\,dt, \qquad \iF{\hat{f}(\omega)} = f(t) = \frac{1}{2\pi} \int \hat{f}(\omega) \exp{-i\omega t}\,d\omega,\\\end{split}\end{equation} and applying this to \eqref{equ:CGLE+DM_O1} we get,\work{\begin{equation}\label{equ:CGLE+DM_O1_Sol}\begin{split}i \pder{q^{(0)}}{x} &+ \Del{d}(x)  \pder{q^{(0)}}{tt} = 0,\\i\pder{\hat{q}^{(0)}}{x} &- \omega^2 \Del{d}(x)  \hat{q}^{(0)} = 0,\\\pder{\hat{q}^{(0)}}{x} &+ i\omega^2\Del{d_r}(x) \hat{q}^{(0)} = 0,\\\pder{\hat{q}^{(0)}}{x} &- \left[ -i \omega^2\Del{d}(x) \right] \hat{q}^{(0)} = 0,\\\pder{\hat{q}^{(0)}}{x} &= \left[ -i \omega^2 \Del{d}(x)\right] \hat{q}^{(0)},\\\hat{q}^{(0)}(z,x,t) &=  \hat{p}(z,\omega) \exp{-i \omega^2 D(x)}\\\mbox{where }D(x) &= \int_{0}^{x} \Del{d}(y) \,dy\\\end{split}\end{equation}}{% \begin{equation}\label{equ:CGLE+DM_O1_Sol}% \begin{split}% \hat{q}^{(0)}(z,x,t) &=  \hat{p}(z,t) \exp{-i \omega^2 \Xi(x)},\\% \mbox{where }\Xi(x) &= \int_{0}^{x} \left(\Del{d_r}(y) - i \Del{d_i}(y)\right) \,dy.\\% \end{split}% \end{equation}}\paragraph{At $\O{x_p}$:}\begin{equation}\begin{split}i &q^{(1)}_x + \Del{d}(x) \pder{q^{(1)}}{tt}\\&= -\left[i q^{(0)}_z + \avg{d} \pder{q^{(0)}}{tt} +s |q^{(0)}|^2 q^{(0)}\right] .\\&\hat{q}^{(1)}_x - \left[-i \omega^2  \Del{d}(x) \right] \hat{q}^{(1)} \\&= i\left[i \hat{q}^{(0)}_z - \omega^2 \left(\avg{d}\right) \hat{q}^{(0)} + s(x)\F{|q^{(0)}|^2 q^{(0)}} \right].\\\end{split}\end{equation}The right-hand side (RHS) of Eqn. above must be orthogonal to the homogeneous solution of adjoint operator of the left-hand side (LHS). Consider the following:\begin{equation}\label{equ:CGLE+DM_O2_1}\begin{split}&\int_{0}^{1} \hat{q}^{(1)\dagger} \left(\hat{q}^{(1)}_x - \left[-i \omega^2  \Del{d}(x)\right] \hat{q}^{(1)}\right) \, dx \\= & \int_{0}^{1} \hat{q}^{(1)\dagger} \hat{q}^{(1)}_x - \left[-i \omega^2  \Del{d}(x)\right]  \hat{q}^{(1)\dagger}  \hat{q}^{(1)} \, dx \\= & \hat{q}^{(1)\dagger} \hat{q}^{(1)} \bigg|_{0}^{1}  - \int_{0}^{1}  \hat{q}^{(1)} \hat{q}^{(1)\dagger}_x \, dx - \int_{0}^{1}\left[-i \omega^2  \Del{d}(x) \right]  \hat{q}^{(1)\dagger}  \hat{q}^{(1)} \, dx \\= & \int_{0}^{1}\hat{q}^{(1)} \left( -\hat{q}^{(1)\dagger}_x - \left[-i \omega^2  \Del{d}(x) \right]  \hat{q}^{(1)\dagger} \right) \, dx \\\end{split}\end{equation}so the LHS must be orthogonal to the solution of\begin{equation}\begin{split}\hat{q}^{(1)\dagger}_x = \left[i \omega^2  \Del{d}(x)\right]  \hat{q}^{(1)\dagger}\\\end{split}\end{equation}which is \begin{equation}\begin{split}\hat{q}^{(1)\dagger} =  \hat{f}(z,\omega) \exp{i \omega^2 D(x)}, \\\end{split}\end{equation}Projecting this against both sides of equation \eqref{equ:CGLE+DM_O2_1} with the innerproduct defined as,\begin{equation}\label{equ:Innerproduct}\begin{split}\left<f(x),g(x)\right> = \int_{0}^{1} f(x)\,g(x) \,dx,\end{split}\end{equation}yields the evolution of the unknown \textit{slow} function $\hat{p}(z,t)$.\begin{equation}\begin{split}i &\left<\exp{i \omega^2 D(x)} \hat{q}^{(0)}_z\right> - \omega^2 \avg{d} \left<\exp{i \omega^2 D(x)} \hat{q}^{(0)}\right> \\&+ \left<\exp{i \omega^2 D(x)} s \F{|q^{(0)}|^2 q^{(0)}}\right> - i \left<\exp{i \omega^2 D(x)}  \hat{q}^{(0)}\right>  = 0,\\\end{split}\end{equation}\begin{equation}\begin{split}i &\hat{p}_z - \omega^2 \avg{d} \hat{p} + s \int_{0}^{1}\exp{i \omega^2 D(x)}  \F{|q^{(0)}|^2 q^{(0)}}\,dx  = 0\\\end{split}\end{equation}\begin{equation}\begin{split}\F{|q^{(0)}|^2 q^{(0)}} &= \int q^{(0)}(z,x,t)  q^{(0)}(z,x,t) \conj{q}^{(0)}(z,x,t) \exp{i\omega t} dt \\&= \int \frac{1}{(2\pi)^3} \iiint \hat{q}^{(0)}(z,x,\omega')  \hat{q}^{(0)}(z,x,\omega'') \conj{\hat{q}}^{(0)}(z,x,\omega^{\prime \prime \prime}) \exp{i(\omega^{\prime \prime \prime}-\omega''-\omega')t} \,d\omega' \,d\omega'' \,d\omega^{\prime \prime \prime} \,\exp{i\omega t} \,dt \\&= \frac{1}{(2\pi)^3} \iiint \hat{q}^{(0)}(z,x,\omega')  \hat{q}^{(0)}(z,x,\omega'') \conj{\hat{q}}^{(0)}(z,x,\omega^{\prime \prime \prime}) \int \exp{i(\omega^{\prime \prime \prime}-\omega''-\omega' +\omega)t} \,dt \,d\omega' \,d\omega'' \,d\omega^{\prime \prime \prime}  \\&= \frac{1}{(2\pi)^2} \iiint \hat{q}^{(0)}(z,x,\omega')  \hat{q}^{(0)}(z,x,\omega'') \conj{\hat{q}}^{(0)}(z,x,\omega^{\prime \prime \prime}) \delta\left(\omega^{\prime \prime \prime}-\omega''-\omega' +\omega \right) \,d\omega' \,d\omega'' \,d\omega^{\prime \prime \prime} \\&= \frac{1}{(2\pi)^2} \iint \hat{q}^{(0)}(z,x,\omega')  \hat{q}^{(0)}(z,x,\omega'') \conj{\hat{q}}^{(0)}(z,x,\omega''+\omega'-\omega)  \,d\omega' \,d\omega'' \\&= \frac{1}{(2\pi)^2} \iint \hat{p}(z,\omega')  \hat{p}(z,\omega'') \conj{\hat{p}}(z,\omega''+\omega'-\omega) \\&\quad \quad \exp{-i \omega'^2 D(x)} \exp{-i \omega''^2 D(x)} \exp{i (\omega''+\omega'-\omega)^2 D(x)} \,d\omega' \,d\omega'' \\&\quad  -\omega'^2 -\omega''^2 + \omega''^2 + 2\omega'' \omega' + \omega'^2  - 2\omega'' \omega - 2\omega' \omega +\omega^2 = 2\omega'' \omega' - 2\omega'' \omega - 2\omega' \omega +\omega^2 \\&\quad  -\omega'^2 -\omega''^2 - \omega''^2 - 2\omega'' \omega' - \omega'^2  + 2\omega'' \omega + 2\omega' \omega - \omega^2 \\&\quad =  -2\omega'^2 - 2\omega''^2 - 2\omega'' \omega' + 2\omega'' \omega + 2\omega' \omega - \omega^2 \\\implies \F{|q^{(0)}|^2 q^{(0)}} &= \frac{1}{(2\pi)^2} \iint \hat{p}(z,\omega')  \hat{p}(z,\omega'') \conj{\hat{p}}(z,\omega''+\omega'-\omega)\\&\quad \quad \exp{i \left( 2\omega'' \omega' - 2\omega'' \omega - 2\omega' \omega +\omega^2\right) D(x)} \,d\omega' \,d\omega'' \\\end{split}\end{equation}and\begin{equation}\begin{split}&\int_{0}^{1}\exp{i \omega^2 D(x)} s \F{|q^{(0)}|^2 q^{(0)}}\,dx \\&= \frac{1}{(2\pi)^2} \iint \hat{p}(z,\omega')  \hat{p}(z,\omega'') \conj{\hat{p}}(z,\omega''+\omega'-\omega) \\&\quad \quad \int_{0}^{1} s \exp{i 2\left( \omega'' \omega' - \omega'' \omega - \omega' \omega + \omega^2\right) D(x)} \,dx \,d\omega' \,d\omega'' \\&\omega'' \omega' - \omega'' \omega - \omega' \omega + \omega^2 = (\omega'-\omega)(\omega''-\omega)\\&\omega'^2 + \omega''^2 + \omega'' \omega' - \omega'' \omega - \omega' \omega = \omega'^2 + \omega''^2 - \omega^2 + (\omega'-\omega)(\omega''-\omega)\\&= \iint \hat{p}(z,\omega')  \hat{p}(z,\omega'') \conj{\hat{p}}(z,\omega''+\omega'-\omega) \hat{K}(\omega',\omega'', \omega)  \,d\omega' \,d\omega'' \\&  \hat{K}(\omega',\omega'',\omega) = \frac{1}{(2\pi)^2}\int_{0}^{1}s \exp{i2(\omega'-\omega)(\omega'' - \omega)D(x)}   \,dx,  \\&\mbox{Let } \omega' =  \omega_1 + \omega, \mbox{ and } \omega'' = \omega_2+\omega\\&= \iint \hat{p}(z, \omega_1 + \omega)  \hat{p}(z,\omega_2+\omega) \conj{\hat{p}}(z,\omega_2+\omega_1 + \omega) \hat{K}(\omega_1,\omega_2,\omega)  \,d\omega_1 \,d\omega_2 \\&  \red{\hat{K}(\omega_1,\omega_2,\omega) = \frac{1}{(2\pi)^2}\int_{0}^{1}s \exp{i2\omega_1\omega_2 D(x)} \,dx,}  \\\end{split}\end{equation}\begin{equation}\begin{split}\hat{K}(\omega_1,\omega_2) = \frac{1}{(2\pi)^2}\int_{0}^{1} s \exp{2\left(i\omega_1\omega_2 D(x) \right) } \,dx,\end{split}\end{equation}Finally, we arrive at what we refer to as DMNLSE in the frequency domain.\\\textbf{DMCGLE Fourier:}\begin{equation}\label{equ:DMCGLE_Fourier}\begin{split}i&\hat{p}_z - \omega^2 \avg{d} \hat{p} +\iint \hat{p}(z,\omega_2+\omega) \hat{p}(z,\omega_1+\omega) \conj{\hat{p}}(z,\omega_1+\omega_2+\omega) \hat{K}(\omega_1,\omega_2)  \,d\omega_1 d\omega_2= 0, \\& \hat{K}(\omega_1,\omega_2) = \frac{1}{(2\pi)^2}\int_{0}^{1} s \exp{2i\omega_1\omega_2 D(x) }  \,dx,  \\\end{split}\end{equation}For the time domain,\begin{equation}\label{equ:CGLE+DM_O2_4}\begin{split}&\iF{ \iint \hat{p}(z, \omega_1 + \omega)  \hat{p}(z,\omega_2+\omega) \conj{\hat{p}}(z,\omega_2+\omega_1 + \omega) \hat{K}(\omega_1,\omega_2)  \,d\omega_1 \,d\omega_2 }\\& = \frac{1}{2\pi} \int \left[\iint \hat{p}(z, \omega_1 + \omega)  \hat{p}(z,\omega_2+\omega) \conj{\hat{p}}(z,\omega_2+\omega_1 + \omega) \hat{K}(\omega_1,\omega_2) \,d\omega_1 \,d\omega_2 \right] \, \exp{-i \omega t} \,d\omega\\& = \frac{1}{2\pi} \int \iint  \left[\iiint p(z, t') p(z,t'') \conj{p}(z,t^{\prime \prime \prime}) \exp{i (\omega_1 + \omega) t'} \exp{i (\omega_2 + \omega) t''} \exp{-i (\omega_1 + \omega_2 + \omega) t^{\prime \prime \prime}} \,dt'dt'' dt^{\prime \prime \prime} \right] \\&\quad \quad \hat{K}(\omega_1,\omega_2) \,d\omega_1 \,d\omega_2 \, \exp{-i \omega t} \,d\omega\\& = \frac{1}{2\pi}  \iiint p(z, t') p(z,t'') \conj{p}(z,t^{\prime \prime \prime}) \iint \exp{i \omega_1 (t'-t^{\prime \prime \prime}) } \exp{i \omega_2 (t''- t^{\prime \prime \prime}) }   \hat{K}(\omega_1,\omega_2) \,d\omega_1 \,d\omega_2  \, \int \exp{ \omega (t' +t'' - t^{\prime \prime \prime} - t) } \,d\omega  \, dt'dt'' dt^{\prime \prime \prime} \\& =  \iiint p(z, t') p(z,t'') \conj{p}(z,t^{\prime \prime \prime}) \iint \exp{i \omega_1 (t'-t^{\prime \prime \prime}) } \exp{i \omega_2 (t''- t^{\prime \prime \prime}) }   \hat{K}(\omega_1,\omega_2) \,d\omega_1 \,d\omega_2  \, \delta(t' +t'' - t^{\prime \prime \prime} - t)   \, dt'dt'' dt^{\prime \prime \prime} \\& =  \iint p(z, t') p(z,t'') \conj{p}(z,t' +t'' - t) \iint \exp{-i \omega_1 (t'' - t) } \exp{-i \omega_2 (t'- t) }   \hat{K}(\omega_1,\omega_2) \,d\omega_1 \,d\omega_2   \, dt'dt''\\& =  \iint p(z, t_1+t) p(z,t_2 + t) \conj{p}(z,t_1 + t_2 + t) K(t_1,t_2)  \, dt_1 dt_2,\quad K(t_1,t_2) = \iint \exp{-i \omega_1 t_2 } \exp{-i \omega_2 t_1 }   \hat{K}(\omega_1,\omega_2) \,d\omega_1 \,d\omega_2  \end{split}\end{equation}\textbf{DMNLSE Time:}\begin{equation}\label{equ:DMCGLE_Time}\begin{split}i&p_z + \left(\breve{d}_r - i \breve{d}_i\right)p_{tt} - i \breve{g} p + \iint p(z,t_2+t) p(z,t_1+t) \conj{p}(z,t_1+t_2+t) K(t_1,t_2)  \,dt_1 dt_2 = 0, \\& K(t_1,t_2) = \iint \exp{-i \omega_1 t_1 } \exp{-i \omega_2 t_2 }   \hat{K}(\omega_1,\omega_2) \,d\omega_1 \,d\omega_2   \\\end{split}\end{equation}where the kernel is given by,\begin{equation}\label{equ:DMCGLE_Kernal}\begin{split}K(t_1,t_2) &= \iint \exp{-i \omega_1 t_1} \exp{-i \omega_2 t_2 }   \hat{K}(\omega_1,\omega_2) \,d\omega_1 \,d\omega_2 \\\hat{K}(\omega_1,\omega_2) &= \frac{1}{(2\pi)^2}\int_{0}^{1}s \exp{2i \omega_1\omega_2 D(x)}  \,dx,  \\\end{split}\end{equation}\begin{equation}\begin{split}D(x)&= \int_{0}^{x} \Delta d(y) \, dy\\\avg{d} &= \int_{0}^{1} \Delta d(x) \,dx = d_{2}(1-\theta_g) + d_{1}\theta_g\\\Delta d(x)&= d(x) - \avg{d} = \left(d_{1}-d_{2}\right) \begin{cases} -\theta_g & 0 < x < \frac{(1- \theta_{g})}{2} \\1-\theta_g & \frac{(1- \theta_{g})}{2} < x < \frac{(1+ \theta_{g})}{2}  \\-\theta_g &  \frac{(1+ \theta_{g})}{2}  < x < 1\end{cases}\\D(x) &= \left(d_{1}-d_{2}\right) \begin{cases} -\theta_g x & 0 < x < \frac{(1- \theta_{g})}{2} \\(1-\theta_g) (x-\frac{1}{2}) & \frac{(1- \theta_{g})}{2} < x < \frac{(1+ \theta_{g})}{2}  \\-\theta_g (x-1) &  \frac{(1+ \theta_{g})}{2}  < x < 1\end{cases}\\D_{str} &= \int_{0}^{1} |\Delta d(x)| \, dx = |d_{1} - d_{2}|\left[2\theta_g\left(1-\theta_g\right)\right]\\\implies D(x) &= \frac{D_{str}}{2}  \begin{cases} - \frac{x}{(1- \theta_{g})} & 0 < x < \frac{(1- \theta_{g})}{2} \\\frac{(x-\frac{1}{2})}{\theta_g}  & \frac{(1- \theta_{g})}{2} < x < \frac{(1+ \theta_{g})}{2}  \\- \frac{(x-1)}{(1- \theta_{g})} &  \frac{(1+ \theta_{g})}{2}  < x < 1\end{cases}\\\end{split}\end{equation}
