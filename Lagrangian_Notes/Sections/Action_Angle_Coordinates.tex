\section{Action Angle Coordinates}


\begin{equation}
\begin{split}
 \dot{E} &=  \ip{U_E}{n(t,z)} \\
 \dot{\Omega} & =   \ip{U_\Omega}{n(t,z)} \\
 \dot{T} &=  2\avg{d} \Omega + \ip{ U_T }{ n(t,z)} \\
 \dot{W} &=  2 \avg{d} C W - \frac{1}{\sqrt{2}} \avg{s} \frac{E}{W} \pder{K_1}{C} + \ip{U_W}{ n(t,z)}\\ 
\dot{C} &= 2 \avg{d} \frac{1}{W^4}  - 2 \avg{d} C^2 + \frac{1}{\sqrt{2}}\avg{s} \frac{E}{W} \pder{K_1}{W} + \ip{U_C}{ n(t,z)}\\
\dot{\phi} &= \avg{d} \Omega^2 - \avg{d} \frac{1}{W^2} + \frac{1}{\sqrt{2}} \avg{s} E K_1 - \frac{1}{4\sqrt{2}} \avg{s} EW \pder{K_1}{W} + 2\avg{s}  \sum_{j=-N/2}^{N/2 - 1} |\alpha_j|^2 + \ip{U_{\phi}}{n(t,z)} \\
\end{split}
\end{equation}
Defining the action  and angle as
\begin{equation}
\begin{split}
H &= C^2 W^2 + \frac{1}{W^2}\\
G &= C W^2\\
W & = \sqrt{\frac{G^2+1}{H}}\\
C & = \frac{G}{H} \left(G^2+1\right)
\end{split}
\end{equation}
\begin{equation}
\begin{split}
H &= C^2 W^2 + \frac{1}{W^2}\\
\dot{H} &= 2CW^2 \dot{C} + 2WC^2 \dot{W} - 2\frac{1}{W^3}\dot{W}\\
\dot{H} &= 2CW^2 \left[2 \avg{d} \frac{1}{W^4}  - 2 \avg{d} C^2 + \frac{1}{\sqrt{2}}\avg{s} \frac{E}{W} \pder{K_1}{W} + \ip{U_C}{ n(t,z)}\right] \\ 
        &  + \left(2WC^2 - 2\frac{1}{W^3} \right)\left[ 2 \avg{d} C W - \frac{1}{\sqrt{2}} \avg{s} \frac{E}{W} \pder{K_1}{C}+\ip{U_W}{ n(t,z)}\right] \\ 
\dot{H} &= 4 \avg{d} \frac{C }{W^2}  - 4 \avg{d} C^3W^2 + \sqrt{2}\avg{s} CW E \pder{K_1}{W} + 2CW^2 \ip{U_C}{ n(t,z)} \\ 
        &  + \left[ 4 \avg{d}W^2C^3 - \sqrt{2} \avg{s}  C^2 E \pder{K_1}{C} + 2WC^2 \ip{U_W}{ n(t,z)}\right] \\ 
        &  - \left[ 4\avg{d} \frac{C}{W^2} - \sqrt{2}\avg{s} \frac{E}{W^4} \pder{K_1}{C}+ 2\frac{1}{W^3}\ip{U_W}{ n(t,z)}\right] \\ 
\dot{H} &=   \sqrt{2}\avg{s} CW E \pder{K_1}{W} - \sqrt{2} \avg{s}  C^2 E \pder{K_1}{C} + \sqrt{2}\avg{s} \frac{E}{W^4} \pder{K_1}{C}\\
&+ 2CW^2 \ip{U_C}{ n(t,z)}  + 2WC^2 \ip{U_W}{ n(t,z)} - 2\frac{1}{W^3}\ip{U_W}{ n(t,z)} \\  \\
\dot{H} &=   \sqrt{2}\avg{s}E\left[ CW \pder{K_1}{W} - \left(C^2  - \frac{1}{W^4}\right) \pder{K_1}{C}\right]+ \ip{U_H}{ n(t,z)}  \\ 
U_H  &= 2CW^2 U_C  + 2WC^2 U_W - 2\frac{1}{W^3} U_W = 2CW^2 U_C  + \left(2WC^2 - 2\frac{1}{W^3}\right) U_W \\ 
\end{split}
\end{equation}

\begin{equation}
\begin{split}
G &= C W^2\\
\dot{G} &= \dot{C} W^2 + 2 C W \dot{W}\\ 
\dot{G} &= W^2\left[2 \avg{d} \frac{1}{W^4}  - 2 \avg{d} C^2 + \frac{1}{\sqrt{2}}\avg{s} \frac{E}{W} \pder{K_1}{W} + \ip{U_C}{ n(t,z)}\right] \\
        &+ 2 C W \left[2 \avg{d} C W - \frac{1}{\sqrt{2}} \avg{s} \frac{E}{W} \pder{K_1}{C} + \ip{U_W}{ n(t,z)}\right]\\
\dot{G} &= 2 \avg{d} H + \frac{\avg{s}}{\sqrt{2}} W E \pder{K_1}{W} + W^2 \ip{U_C}{ n(t,z)} - \sqrt{2}\avg{s}  C E \pder{K_1}{C} + 2 C W  \ip{U_W}{ n(t,z)}\\
\dot{G} &= 2 \avg{d} H + \frac{\avg{s}}{\sqrt{2}} W E \pder{K_1}{W}  - \sqrt{2}\avg{s}  C E \pder{K_1}{C} + 2 C W \ip{U_W}{ n(t,z)} + W^2 \ip{U_C}{ n(t,z)}\\
\dot{G} &= 2 \avg{d} H + \frac{\avg{s}}{\sqrt{2}} W E \pder{K_1}{W}  - \sqrt{2}\avg{s}  C E \pder{K_1}{C} + \ip{U_G}{ n(t,z)}\\
\dot{G} &= 2 \avg{d} H + \avg{s}\sqrt{2} E \left[ \frac{1}{2} W \pder{K_1}{W}  - C \pder{K_1}{C}\right] + \ip{U_G}{ n(t,z)}\\
U_G &= 2 C W U_W + W^2 U_C
\end{split}
\end{equation}

\section{Exact Solution}
Fist notice that without the noise terms,
\begin{equation}
\begin{split}
 E(z) &=  E_0 \\
 \Omega(z) & =  \Omega_0 \\
 T(z) &=  2\avg{d} \Omega_0 z \\
 \dot{\phi} &= \avg{d} \Omega_0  + \frac{1}{\sqrt{2}} \avg{s} E_0 \left(K_1 - \frac{1}{4}  W \pder{K_1}{W}\right) - \avg{d} \frac{1}{W^2} \\
\end{split}
\end{equation}

\begin{equation}
\begin{split}
\pder{K_1}{W}  &=  \pder{K_1}{H} \pder{H}{W} + \pder{K_1}{G} \pder{G}{W}\\
 CW \pder{K_1}{W} - \left(C^2  - \frac{1}{W^4}\right) \pder{K_1}{C} &= CW \left( \pder{K_1}{H} \pder{H}{W} + \pder{K_1}{G} \pder{G}{W}\right) - \left(C^2  - \frac{1}{W^4}\right) \left( \pder{K_1}{H} \pder{H}{W} + \pder{K_1}{G} \pder{G}{W}\right) = \\
&= \left[CW \pder{H}{W} -  \left(C^2  - \frac{1}{W^4}\right) \pder{H}{C}\right] \pder{K_1}{H}  +  \left[CW \pder{G}{W} -  \left(C^2  - \frac{1}{W^4}\right) \pder{G}{C}\right] \pder{K_1}{G} \\
&= \left[CW \left(2C^2W-\frac{2}{W^3}\right) -  \left(C^2  - \frac{1}{W^4}\right)2CW^2 \right] \pder{K_1}{H}  +  \left[CW (2CW) - \left(C^2  - \frac{1}{W^4}\right) W^2\right] \pder{K_1}{G} \\
&=  \left[2C^2W^2 - \left(C^2W^2  - \frac{1}{W^2}\right) \right] \pder{K_1}{G} \\
&=  \left[C^2W^2 + \frac{1}{W^2}\right] \pder{K_1}{G} \\
& = H \pder{K_1}{G}
\end{split}
\end{equation}
\begin{equation}
\begin{split}
\pder{K_1}{W}  &=  \pder{K_1}{H} \pder{H}{W} + \pder{K_1}{G} \pder{G}{W}\\
\frac{1}{2} W \pder{K_1}{W}  - C \pder{K_1}{C} &=  \frac{1}{2} W  \left( \pder{K_1}{H} \pder{H}{W} + \pder{K_1}{G} \pder{G}{W}\right) - C \left( \pder{K_1}{H} \pder{H}{W} + \pder{K_1}{G} \pder{G}{W}\right) \\
&= \left[\frac{1}{2} W \pder{H}{W} - C \pder{H}{C}\right] \pder{K_1}{H}  +  \left[\frac{1}{2} W  \pder{G}{W} -  C \pder{G}{C}\right] \pder{K_1}{G} \\
&= \left[\frac{1}{2} W \left( 2 C^2 W - \frac{2}{W^3}\right) - C \left( 2 C W^2 \right) \right] \pder{K_1}{H}  +  \left[\frac{1}{2} W  \left( 2 C W \right) -  C \left(  W^2 \right)\right] \pder{K_1}{G} \\
&= -H \pder{K_1}{H}  \\
\end{split}
\end{equation}



\begin{equation}
\begin{split}
\dot{H} &=   \sqrt{2}\avg{s}E H \pder{K_1}{G}  \\ 
\dot{G} &= 2 \avg{d} H - \avg{s}\sqrt{2} E H \pder{K_1}{H} \\
\end{split}
\end{equation}

Now define $L = \ln{H}$, then $\pder{K_1}{H} = \pder{K_1}{L}\pder{L}{H} = \frac{1}{H} \pder{K_1}{L}$
\begin{equation}
\begin{split}

\end{split}
\end{equation}
