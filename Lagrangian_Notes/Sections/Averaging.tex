\section{Action Angle Coordinates}


\begin{equation}
\begin{split}
 \dot{E} &=  \ip{U_E}{n(t,z)} \\
 \dot{\Omega} & =   \ip{U_\Omega}{n(t,z)} \\
 \dot{T} &=  2\avg{d} \Omega + \ip{ U_T }{ n(t,z)} \\
 \dot{W} &=  2 \avg{d} C W - \frac{1}{\sqrt{2}} \avg{s} \frac{E}{W} \pder{K_1}{C} + \ip{U_W}{ n(t,z)}\\ 
\dot{C} &= 2 \avg{d} \frac{1}{W^4}  - 2 \avg{d} C^2 + \frac{1}{\sqrt{2}}\avg{s} \frac{E}{W} \pder{K_1}{W} + \ip{U_C}{ n(t,z)}\\
\dot{\phi} &= \avg{d} \Omega^2 - \avg{d} \frac{1}{W^2} + \frac{1}{\sqrt{2}} \avg{s} E K_1 - \frac{1}{4\sqrt{2}} \avg{s} EW \pder{K_1}{W} + 2\avg{s}  \sum_{j=-N/2}^{N/2 - 1} |\alpha_j|^2 + \ip{U_{\phi}}{n(t,z)} \\
\end{split}
\end{equation}
Defining the action $H$ and angle $G$ as
\begin{equation}
\begin{split}
H &= C^2 W^2 + \frac{1}{W^2}\\
\dot{H} &= 2CW^2 \dot{C} + 2WC^2 \dot{W} - 2\frac{1}{W^3}\dot{W}\\
\dot{H} &= 2CW^2 \left[2d_{s} \left(\frac{1}{W^4}-C^2\right)-\frac{n_{l}}{\sqrt{2}}  \frac{E}{W^3} - \frac{4\tau}{1+\frac{E \sqrt{\pi}}{E_{sat}}} \frac{C}{W^2} + F_C(z)\right] \\ 
        &  + \left(2WC^2 - 2\frac{1}{W^3} \right)\left[ 2d_{s} C W + l \left[l_0(E,W) - 2l_2(E,W)\right] \frac{W}{\sqrt{\pi}} +  \frac{\tau(z)}{1+\frac{E \sqrt{\pi}}{E_{sat}}} \left(\frac{1}{W} - C^2W^3\right) + F_W(z)\right] \\ 
\dot{H} &= -\sqrt{2}n_{l}E \left(\frac{H}{G^2+1}\right)^{\frac{2}{3}} G + \frac{2l}{\sqrt{\pi}} \left(\frac{H}{G^2+1}\right) \left(G^2-1\right)\left[p_0(E,G,H) - 2p_2(E,G,H)\right] \\
        &  - \frac{2\tau}{1+\frac{E \sqrt{\pi}}{E_{sat}}} H^2 + F_H(z) \\
        F_H(z) &= \left[ 2G F_C(z) + 2\left(\frac{H}{G^2+1}\right)^{\frac{3}{2}} (G^2-1) F_W(z) \right] \\ 
        \langle F_H(z) F_H(z') \rangle &= \frac{\sigma^2}{\sqrt{\pi}} \frac{4}{E}H^2 \delta(z-z') \\ 
\end{split}
\end{equation}

\begin{equation}
\begin{split}
G &= C W^2\\
\dot{G} &= \dot{C} W^2 + 2 C W \dot{W}\\ 
\dot{G} &= W^2\left[2d_{s} \left(\frac{1}{W^4}-C^2\right)-\frac{n_{l}}{\sqrt{2}}  \frac{E}{W^3} - \frac{4\tau}{1+\frac{E \sqrt{\pi}}{E_{sat}}} \frac{C}{W^2} + F_C(z)\right] \\
        &+ 2 C W \left[2d_{s} C W + l \left[p_0(E,W) - 2p_2(E,W)\right] \frac{W}{\sqrt{\pi}} +  \frac{\tau}{1+\frac{E \sqrt{\pi}}{E_{sat}}} \left(\frac{1}{W} - C^2W^3\right) + F_W(z)\right]\\
\dot{G} &= 2 d_s(x) H - \frac{n_{l}}{\sqrt{2}}E \left(\frac{H}{G^2+1}\right)^{\frac{1}{2}} - 2\tau\frac{1}{1+\frac{E \sqrt{\pi}}{E_{sat}}} HG + \frac{2l}{\sqrt{\pi}}\left[p_0(E,G,H) - 2p_2(E,G,H)\right]G  + F_G(z)\\
 F_G(z) &= \frac{G^2+1}{H} F_C(z) + 2 \left(\frac{H}{G^2+1}\right)^\frac{1}{2} G F_W(z)\\
        \langle F_G(z) F_G(z') \rangle &= \frac{\sigma^2}{\sqrt{\pi}} \frac{4}{E}(1+G^2) \delta(z-z') \\ 
\end{split}
\end{equation}

\begin{equation}
\begin{split}
\langle F_H(z) F_G(z') \rangle &= \frac{\sigma^2}{\sqrt{\pi}} \frac{4}{E}HG \delta(z-z') \\ 
\end{split}
\end{equation}
\begin{equation}
\begin{split}

\end{split}
\end{equation}
\begin{equation}
\begin{split}

\end{split}
\end{equation}
\begin{equation}
\begin{split}

\dot{E} &= 2g \frac{1}{1+\frac{E\sqrt{\pi}}{E_{sat}}} E - \frac{2l}{\sqrt{\pi}} p_0(E,G,H) E - \tau \frac{1}{1+\frac{E\sqrt{\pi}}{E_{sat}}} E \left[2\Omega^2 + H\right] + F_E 
\end{split}
\end{equation}

\begin{equation}

\end{split}
\end{equation}

Recalling that:

\begin{equation}
\begin{split}
D_s(x) &= \int_{0}^{x} \Delta d_s(y) \, dy\\
\avg{d}_s &= \int_{0}^{1} d_s(x) \,dx = d_{2} (1-\theta_g) + d_{1} \theta_g\\
\Delta d_s(x) &= d_s(x) - \avg{d}_s = \left(d_{1}-d_{2}\right) 
\begin{cases} -\theta_g & 0 < x < \frac{(1- \theta_{g})}{2} \\
1-\theta_g & \frac{(1- \theta_{g})}{2} < x < \frac{(1+ \theta_{g})}{2}  \\
-\theta_g &  \frac{(1+ \theta_{g})}{2}  < x < 1\end{cases}\\
D_{s,str} &= \int_{0}^{1} |\Delta d_s(x)| \, dx = |d_{1} - d_{2}|\left[2\theta_g\left(1-\theta_g\right)\right]\\
\Delta d_s(x) &= \frac{D_{s,str}}{2}
\begin{cases} -\frac{1}{\left(1-\theta_g\right)} & 0 < x < \frac{(1- \theta_{g})}{2} \\
\frac{1}{\theta_g} & \frac{(1- \theta_{g})}{2} < x < \frac{(1+ \theta_{g})}{2}  \\
-\frac{1}{\left(1-\theta_g\right)} &  \frac{(1+ \theta_{g})}{2}  < x < 1\end{cases}\\
D_s(x) &= \left(d_{1}-d_{2}\right) \begin{cases} -\theta_g x & 0 < x < \frac{(1- \theta_{g})}{2} \\
(1-\theta_g) (x-\frac{1}{2}) & \frac{(1- \theta_{g})}{2} < x < \frac{(1+ \theta_{g})}{2}  \\
-\theta_g (x-1) &  \frac{(1+ \theta_{g})}{2}  < x < 1\end{cases}\\
\implies D_s(x) &= \frac{D_{s,str}}{2}  \begin{cases} - \frac{x}{(1- \theta_{g})} & 0 < x < \frac{(1- \theta_{g})}{2} \\
\frac{(x-\frac{1}{2})}{\theta_g}  & \frac{(1- \theta_{g})}{2} < x < \frac{(1+ \theta_{g})}{2}  \\
- \frac{(x-1)}{(1- \theta_{g})} &  \frac{(1+ \theta_{g})}{2}  < x < 1\end{cases}
\end{split}
\end{equation}
the integral in the $\phi$ equation can be written as:
\begin{equation}
\begin{split}

 &= \frac{1}{2\tilde{H}_0(z)} \begin{cases} 
\atan{\tilde{G}_0(z) - \frac{D_{s,str}}{(1-\theta_g)}\tilde{H}_0(z) x} -\atan{\tilde{G}_0(z)}  & 0 < x < \frac{(1- \theta_{g})}{2} \\
\atan{\tilde{G}_0(z) + \frac{D_{s,str}}{\theta_g}\tilde{H}_0(z) (x-\frac{1}{2})} -\atan{\tilde{G}_0(z)}    & \frac{(1- \theta_{g})}{2} < x < \frac{(1+ \theta_{g})}{2}  \\
\atan{\tilde{G}_0(z) - \frac{D_{s,str}}{(1-\theta_g)}\tilde{H}_0(z) (x-1)} -\atan{\tilde{G}_0(z)} &  \frac{(1+ \theta_{g})}{2}  < x < 1
\end{cases}\\
 &= -\frac{1}{2\tilde{H}_0} \begin{cases}
 \atan{\frac{\frac{D_{s,str}}{(1-\theta_g)}\tilde{H}_0 x}{1+\left(\tilde{G}_0^2-\frac{D_{s,str}}{(1-\theta_g)}\tilde{H}_0\tilde{G}_0 x\right)}}  & 0 < x < \frac{(1- \theta_{g})}{2} \\
-\atan{\frac{\frac{D_{s,str}}{\theta_g}\tilde{H}_0 (x-\frac{1}{2})}{1+\left(\tilde{G}_0^2+\frac{D_{s,str}}{\theta_g}\tilde{H}_0\tilde{G}_0 (x-\frac{1}{2})\right)}}    & \frac{(1- \theta_{g})}{2} < x < \frac{(1+ \theta_{g})}{2}  \\
\atan{\frac{\frac{D_{s,str}}{(1-\theta_g)}\tilde{H}_0 (x-1)}{1+\left(\tilde{G}_0^2-\frac{D_{s,str}}{(1-\theta_g)}\tilde{H}_0\tilde{G}_0 (x-1)\right)}}    &  \frac{(1+ \theta_{g})}{2}  < x < 1 
\end{cases}
\end{split}
\end{equation}
